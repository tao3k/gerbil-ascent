"""Observe compiler/test interference while running the unchanged shape matrix."""
import json
import subprocess
import sys
import threading
import time
from pathlib import Path
# Resolve the checkout directly; receipt directories never choose a second checkout.
root=Path('/Users/guangtao/ghq/github.com/tao3k/gerbil-ascent')
out=root/'rust/benchmarks/results-hybrid-20261015'
checks=[]
stop=threading.Event()
forbidden={'cargo','cargo-clippy','rustc','clippy-driver','clang','clang++','gcc','g++','cc','c++','gsc','gsc-boot','gsc-script','gxi','java','quint','ld','rust-lld','swift-frontend','lean','lake','make','gmake','ninja','cmake','maturin','gxc','gerbil','gxtest'}
def snapshot():
    run=subprocess.run(['/bin/ps','-axo','pid=,ppid=,comm='],capture_output=True,text=True,check=True,timeout=5)
    processes={}
    for line in run.stdout.splitlines():
        pid, parent, command=line.strip().split(maxsplit=2)
        processes[int(pid)]={'pid':int(pid),'parent':int(parent),'command':command}
    busy=[]
    for pid, process in processes.items():
        if Path(process['command']).name not in forbidden:continue
        ancestors=[]
        parent=process['parent']
        for _ in range(8):
            if parent not in processes:break
            ancestors.append(processes[parent])
            parent=processes[parent]['parent']
        busy.append(dict(process,ancestors=ancestors))
    checks.append({'seconds':time.monotonic(),'busy':busy})
    return busy
def monitor():
    while not stop.wait(.5):
        try: snapshot()
        except Exception as error: checks.append({'error':str(error)})
for _ in range(10):
    if snapshot():
        raise SystemExit('active compiler/test process; benchmark not started: '+str(checks[-1]))
    time.sleep(.5)
watch=threading.Thread(target=monitor);watch.start()
try:
    result=subprocess.run([sys.executable,str(root/'rust/benchmarks/run-hybrid.py'),'--binary',str(root/'rust/target/debug/hybrid-benchmark'),'--output',str(out)],cwd=root)
finally:
    stop.set();watch.join();snapshot()
    if out.is_dir():
        receipt={'interval_seconds':.5,'checks':checks,'build_isolation_passed':all(not c.get('busy') and 'error' not in c for c in checks)}
        (out/'process-audit.json').write_text(json.dumps(receipt,indent=2)+'\n')
if result.returncode:raise SystemExit(result.returncode)
if any(c.get('busy') or 'error' in c for c in checks):raise SystemExit('concurrent build/test interference observed; timings diagnostic only')
